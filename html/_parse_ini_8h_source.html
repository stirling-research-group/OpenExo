122 enum class joint_id : uint8_t //Joint IDs
123 {
124 //Byte format : [arm_2, is_left, !is_left, arm_1, is_elbow, is_ankle, is_knee, is_hip]
125 left = 0b01000000,
126 right = 0b00100000,
127
128 hip = 0b00000001,
129 knee = 0b00000010,
130 ankle = 0b00000100,
131 elbow = 0b00001000,
132 arm_1 = 0b00010000,
133 arm_2 = 0b10000000,
134
135 left_hip = left|hip, //Set Motor ID to: 65
136 left_knee = left|knee, //Set Motor ID to: 66
137 left_ankle = left|ankle, //Set Motor ID to: 68
138 left_elbow = left|elbow, //Set Motor ID to: 72
139 left_arm_1 = left|arm_1, //Set Motor ID to: 80
140 left_arm_2 = left|arm_2, //Set Motor ID to: 192
141 //Unused Joint ID: 96
142
143 right_hip = right|hip, //Set Motor ID to: 33
144 right_knee = right|knee, //Set Motor ID to: 34
145 right_ankle = right|ankle, //Set Motor ID to: 36
146 right_elbow = right|elbow, //Set Motor ID to: 40
147 right_arm_1 = right|arm_1, //Set Motor ID to: 48
148 right_arm_2 = right|arm_2, //Set Motor ID to: 160
149 //Unused Joint ID: 56
150 };
