132 inline static void get_controller_params(UARTHandler *handler, ExoData *exo_data, UART_msg_t msg)
133 {
134 //logger::println("UART_command_handlers::update_controller_params->Fetching params with msg: ");
135 //UART_msg_t_utils::print_msg(msg);
136
137 JointData *j_data = exo_data->get_joint_with(msg.joint_id);
138 if (j_data == NULL)
139 {
140 //logger::println("UART_command_handlers::get_controller_params->No joint with id = ");
141 //logger::print(msg.joint_id);
142 //logger::println(" found");
143 return;
144 }
145
146 msg.command = UART_command_names::update_controller_params;
147
148 uint8_t param_length = j_data->controller.get_parameter_length();
149 msg.len = param_length + (uint8_t)UART_command_enums::controller_params::LENGTH;
150 msg.data[(uint8_t)UART_command_enums::controller_params::CONTROLLER_ID] = j_data->controller.controller;
151 msg.data[(uint8_t)UART_command_enums::controller_params::PARAM_LENGTH] = param_length;
152 for (int i = 0; i < param_length; i++)
153 {
154 msg.data[(uint8_t)UART_command_enums::controller_params::PARAM_START + i] = j_data->controller.parameters[i];
155 }
156
157 handler->UART_msg(msg);
158 }

689 {
690 UART_msg_t rx_msg = {0, 0, 0, 0};
691 uint8_t searching = 1;
692 // logger::println("UART_command_utils::call_and_response->searching for message");
693 float start_time = millis();
694 while (searching)
695 {
696 handler->UART_msg(msg);
697 // logger::println("UART_command_utils::call_and_response->sent msg");
698 delay(500);
699 rx_msg = handler->poll(200000);
700 searching = (rx_msg.command != (msg.command + 1));
701
702 if (millis() - start_time > timeout)
703 {
704 // logger::println("UART_command_utils::call_and_response->timed out");
705 return rx_msg;
706 }
707 }
708 // logger::println("UART_command_utils::call_and_response->found message:");
709 UART_msg_t_utils::print_msg(rx_msg);
710 return rx_msg;

713 static uint8_t get_config(UARTHandler *handler, uint8_t *config, float timeout)
714 {
715 UART_msg_t msg;
716 float start_time = millis();
717 while (1)
718 {
719 msg.command = UART_command_names::get_config;
720 msg.len = 0;
721 msg = call_and_response(handler, msg, timeout);
722
723 if ((millis() - start_time) > timeout)
724 {
725 logger::println("UART_command_utils::get_config->timed out");
726 return 1;
727 }
728
729 //The length of the message needs to be equal to the config length
730 if (msg.len != ini_config::number_of_keys)
731 {
732 logger::println("UART_command_utils::get_config->msg.len != number_of_keys");
733 //Keep trying to get config
734 continue;
735 }
736 for (int i = 0; i < msg.len; i++)
737 {
738 //A valid config will not contain a zero
739 if (!msg.data[i])
740 {
741 logger::print("UART_command_utils::get_config->Config contained a zero at index ");
742 logger::println(i);
743
744 //Keep trying to get config
745 continue;
746 }
747 }
748 logger::println("UART_command_utils::get_config->got good config");
749 break;
750 }
751
752 //Pack config
753 for (int i = 0; i < msg.len; i++)
754 {
755 config[i] = msg.data[i];
756 }
757 return 0;
758 }
